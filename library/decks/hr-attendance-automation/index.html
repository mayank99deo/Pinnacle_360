HR DATA AUTOMATION · 02 SOLUTION

Apps Script syncs HROne attendance into the sheet

set up once ↓
sync.gs · Apps ScriptJavaScript
function syncAttendance() {
const key = getSecret('HRONE_KEY');
const res = UrlFetchApp.fetch(
HRONE + '/attendance?date=' + today,
{ headers: { Authorization: key } });
writeToSheet(JSON.parse(res));
}
// one-time: install the trigger
ScriptApp.newTrigger('syncAttendance')
.timeBased().everyDays(1)
.atHour(10).create();
